01 پایه‌های Arduino

Digital

پایه دیجیتال دو وضعیت اصلی دارد: HIGH و LOW. برای LED، کلید و سنسورهای دیجیتال استفاده می‌شود.

Analog

در Arduino UNO تابع analogRead معمولاً عددی بین 0 تا 1023 برمی‌گرداند.

PWM

با analogWrite می‌توان PWM ایجاد کرد و مثلاً سرعت موتور را کنترل کرد.

INPUT_PULLUP

برای کلیدها بسیار کاربردی است و نیاز به مقاومت Pull-up خارجی را در بسیاری از مدارهای ساده حذف می‌کند.

اولین برنامه


const int LED = 13;

void setup() {
  pinMode(LED, OUTPUT);
}

void loop() {

  digitalWrite(LED, HIGH);
  delay(500);

  digitalWrite(LED, LOW);
  delay(500);
}

02 دستورات مهم

دستور کاربرد
pinMode() تعیین ورودی یا خروجی
digitalRead() خواندن دیجیتال
digitalWrite() نوشتن دیجیتال
analogRead() خواندن آنالوگ
analogWrite() PWM
millis() زمان‌سنجی بدون توقف برنامه
constrain() محدود کردن مقدار

millis


unsigned long lastTime = 0;

const unsigned long interval = 100;

void loop() {

  if (millis() - lastTime >= interval) {

    lastTime = millis();

    // کار دوره‌ای

  }

}

03 سنسورها

IR

برای تشخیص خط و سطح استفاده می‌شود.

Ultrasonic

برای اندازه‌گیری فاصله با امواج صوتی.

LDR

برای اندازه‌گیری شدت نور.

Encoder

برای اندازه‌گیری حرکت و سرعت چرخ.

HC-SR04


const int TRIG = 8;
const int ECHO = 9;

long getDistance() {

  digitalWrite(TRIG, LOW);
  delayMicroseconds(2);

  digitalWrite(TRIG, HIGH);
  delayMicroseconds(10);

  digitalWrite(TRIG, LOW);

  long time = pulseIn(ECHO, HIGH, 30000);

  if(time == 0)
    return -1;

  return time / 58;
}

void setup() {

  pinMode(TRIG, OUTPUT);
  pinMode(ECHO, INPUT);

  Serial.begin(9600);
}

void loop() {

  Serial.println(getDistance());

  delay(100);
}

04 موتور DC

موتور DC را مستقیماً به پایه Arduino وصل نکن. برای کنترل موتور از Motor Driver استفاده کن.

⚠️ موتور جریان زیادی نیاز دارد و می‌تواند باعث افت ولتاژ یا ریست Arduino شود.

05 درایور موتور

درایور ویژگی
L298N رایج و ساده، اما نسبتاً پراتلاف
TB6612FNG مناسب ربات‌های کوچک و کم‌اتلاف‌تر
L293D قدیمی‌تر و مناسب بارهای سبک
GND سیستم کنترل و درایور باید در مدار معمولی مشترک باشد.

07 Line Follower

فرمول ذهنی ساده:

سنسور ← تشخیص موقعیت خط ← محاسبه خطا ← اصلاح موتور ← تکرار

int C = digitalRead(3);
int R = digitalRead(4);

if(C && !L && !R) {

  // مستقیم

}

else if(L) {

  // چپ

}

else if(R) {

  // راست

}

else {

  // خط پیدا نشد

}
مقدار HIGH و LOW به نوع سنسور و نحوه سیم‌بندی بستگی دارد.

08 PID

P

با افزایش خطا، مقدار اصلاح افزایش پیدا می‌کند.

I

خطاهای گذشته را جمع می‌کند.

D

سرعت تغییر خطا را بررسی می‌کند. 


float Kp = 25;
float Ki = 0;
float Kd = 12;

float integral = 0;
float previousError = 0;

float PID(float error, float dt) {

  integral += error * dt;

  float derivative =
      (error - previousError) / dt;

  previousError = error;

  return
    Kp * error +
    Ki * integral +
    Kd * derivative;
}
⚠️ ضرایب Kp و Ki و Kd باید با ربات واقعی تنظیم شوند.
